# 编码规则（单号生成器）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 控制台系统管理新增「编码规则」配置功能，管理员可视化配置单号规则（9 类明细可拖拽排序），系统内部各服务经 facade 单个/批量生成全局唯一单号。

**Architecture:** 规则与明细存 `mdc_code_rule`/`mdc_code_rule_item`（明细差异配置 JSON 列）；生成引擎为"流水线 + 策略模式"（每个规则类别一个 `RuleItemGenerator`）；流水号用 RedisScript（Lua）原子"初始化+递增"，周期偏移只作用于 periodKey，DB 水位表冷备恢复（跳跃前进不重复）；内部调用经 console-facade（boot 本地 / cloud `/inner/**` Feign）。

**Tech Stack:** Java 17 + Spring Boot 3 + MyBatis-Flex + Lombok + JUnit5 + StringRedisTemplate(RedisScript)；Vue3 + vben（vxe-table 行拖拽、useVbenModal、useVbenDrawer）

**Spec:** `docs/superpowers/specs/2026-10-28-code-rule-design.md`

## Global Constraints

- **禁止任何 git add / git commit / git push 操作**
- 实体分层：`entity/base/XxxBase.java` 只放表字段；实体继承 `SuperEntity<Long>`（id、created_at、created_by、updated_at、updated_by）
- Mapper 接口必须加 `@Repository` 注解
- Service 继承：`XxxService extends SuperService<Xxx>`、`XxxServiceImpl extends SuperServiceImpl<XxxMapper, Xxx>`
- 统一返回 `R<T>`；分页入参 `PageParams<XxxQuery>`、出参 `R<Page<XxxVo>>`
- JSON 配置列统一 camelCase 字段 + `@Column(typeHandler = Fastjson2TypeHandler.class)`（参照 InterfaceConfigBase.configJson）

- 规则/明细缓存走 `cacheOps.get(CacheKey, loader)`（参照 ConfigServiceImpl.getByUniqKey），失效用 `cacheOps.del(key)`
- 所有弹窗/抽屉必须用 `useVbenModal` / `useVbenDrawer`；行拖拽参照 `web-console/.../interfaceConfig/data/config-json.tsx` 的 `rowDragConfig` 用法

```java
import io.swagger.v3.oas.annotations.media.Schema;

import lombok.Getter;

/**

@Getter
@AllArgsConstructor
@Schema(title = "CodeRuleTypeEnum", description = "编码规则-规则类别")

    /** 常量 */
    CONSTANT(10, "常量"),
    /** 日期 */
    DATE(20, "日期"),
    /** 变量 */
    VARIABLE(30, "变量"),
    /** 流水号 */
    SERIAL(40, "流水号"),
```

```java
package top.mddata.console.service.coderule.support;

import top.mddata.base.utils.ArgumentAssert;
import top.mddata.console.enumeration.coderule.OffsetUnitEnum;
import top.mddata.console.enumeration.coderule.ResetRuleEnum;

import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.time.temporal.IsoFields;

/**
 * 流水号周期标签计算器。
 * 周期偏移只作用于本类（periodKey 生成），不影响日期明细的取值。
 */
public class PeriodKeyResolver {

    private PeriodKeyResolver() {
    }

    /**
     * 计算周期标签
     *
     * @param resetRule 归零规则
     * @param cycleOffset 周期偏移量（0=自然周期）
     * @param offsetUnit 偏移单位
     * @param bizTime 生成时刻
     * @return 周期标签
     */
    public static String resolve(ResetRuleEnum resetRule, Integer cycleOffset, OffsetUnitEnum offsetUnit, LocalDateTime bizTime) {
        ArgumentAssert.notNull(resetRule, "流水号规则必须配置归零规则");
        if (resetRule == ResetRuleEnum.NEVER) {
            return "FOREVER";
        }
        LocalDateTime baseTime = shift(bizTime, cycleOffset == null ? 0 : cycleOffset, offsetUnit);
        return switch (resetRule) {
            case DAY -> baseTime.format(DateTimeFormatter.ofPattern("yyyyMMdd"));
            case MONTH -> baseTime.format(DateTimeFormatter.ofPattern("yyyyMM"));
            case YEAR -> baseTime.format(DateTimeFormatter.ofPattern("yyyy"));
            case WEEK -> baseTime.get(IsoFields.WEEK_BASED_YEAR) + "W"
                    + String.format("%02d", baseTime.get(IsoFields.WEEK_OF_WEEK_BASED_YEAR));
            default -> throw new IllegalArgumentException("不支持的归零规则: " + resetRule);
        };
    }

    private static LocalDateTime shift(LocalDateTime bizTime, int cycleOffset, OffsetUnitEnum offsetUnit) {
        if (cycleOffset == 0) {
            return bizTime;
        }
        // 默认按小时，页面只暴露 HOUR/DAY
        return (offsetUnit == OffsetUnitEnum.DAY) ? bizTime.minusDays(cycleOffset) : bizTime.minusHours(cycleOffset);
    }
}
```

```java
package top.mddata.console.service.coderule.support;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.data.redis.core.script.RedisScript;
import org.springframework.stereotype.Component;
import top.mddata.console.entity.coderule.CodeSn;
import top.mddata.console.mapper.coderule.CodeSnMapper;
import com.mybatisflex.core.query.QueryWrapper;

import java.util.ArrayList;
import java.util.List;

/**
 * 流水号分配器：Redis 原子递增（Lua 一次性完成 初始化+递增+TTL），DB 水位表冷备恢复。
 * 无锁、集群安全；恢复时水位跳跃前进（允许跳号，绝不重复）。
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class SnAllocator {

    /** 水位回写间隔：Redis 值每跨 1000 回写一次 DB */
    private static final long WATER_STEP = 1000L;

    /**
     * Lua：key 不存在时用 ARGV[1] 初始化（并设置 TTL，ARGV[3]<=0 不过期），然后 INCRBY ARGV[2]。
     * 初始化值由调用方算好：新周期=initValue-step；冷恢复=DB水位+1000。
     */
    private static final RedisScript<Long> INIT_AND_INCR = RedisScript.of("""
            if redis.call('exists', KEYS[1]) == 0 then
                redis.call('set', KEYS[1], ARGV[1])
                local ttl = tonumber(ARGV[3])
                if ttl > 0 then redis.call('pexpire', KEYS[1], ttl) end
            end
            return redis.call('incrby', KEYS[1], ARGV[2])
            """, Long.class);

    private final StringRedisTemplate stringRedisTemplate;
    private final CodeSnMapper codeSnMapper;

    /**
     * 原子分配 count 个连续流水号
     *
     * @param ruleItemId 明细ID
     * @param periodKey 周期标签
     * @param initValue 初始值
     * @param step 递增值
     * @param count 分配数量
     * @param ttlMillis Redis key 的 TTL（<=0 不过期）
     * @return 连续递增值列表（首个为本次分配的起始值）
     */
    public List<Long> allocate(Long ruleItemId, String periodKey, long initValue, long step, int count, long ttlMillis) {
        String key = "mdc:sn:" + ruleItemId + ":" + periodKey;
        long initBase = resolveInitBase(ruleItemId, periodKey, initValue, step);
        Long last = stringRedisTemplate.execute(INIT_AND_INCR, List.of(key),
                String.valueOf(initBase), String.valueOf(step * (long) count), String.valueOf(ttlMillis));
        if (last == null) {
            throw new IllegalStateException("流水号分配失败: " + key);
        }
        writeBackWaterMark(ruleItemId, periodKey, last);
        List<Long> result = new ArrayList<>(count);
        long first = last - step * (long) count + step;
        for (int i = 0; i < count; i++) {
            result.add(first + step * i);
        }
        return result;
    }

    /**
     * 计算 Redis 初始化基数（仅在 key 不存在时生效）：
     * DB 有水位 → 水位+1000（冷恢复跳跃，绝不重复）；无水位 → initValue-step（首个值正好为 initValue）
     */
    private long resolveInitBase(Long ruleItemId, String periodKey, long initValue, long step) {
        CodeSn sn = codeSnMapper.selectOneByQuery(QueryWrapper.create()
                .eq(CodeSn::getRuleItemId, ruleItemId).eq(CodeSn::getPeriodKey, periodKey));
        return sn != null ? sn.getCurrentValue() + WATER_STEP : initValue - step;
    }

    /**
     * 水位回写：Redis 值每跨过 1000 的整数倍时更新 DB（upsert）
     */
    private void writeBackWaterMark(Long ruleItemId, String periodKey, long current) {
        if (current % WATER_STEP != 0) {
            return;
        }
        CodeSn sn = codeSnMapper.selectOneByQuery(QueryWrapper.create()
                .eq(CodeSn::getRuleItemId, ruleItemId).eq(CodeSn::getPeriodKey, periodKey));
        if (sn == null) {
            sn = new CodeSn().setRuleItemId(ruleItemId).setPeriodKey(periodKey).setCurrentValue(current);
            codeSnMapper.insert(sn);
        } else {
            CodeSn update = new CodeSn();
            update.setId(sn.getId());
            update.setCurrentValue(current);
            codeSnMapper.update(update);
        }
    }
}
```

```java
package top.mddata.console.service.coderule.generator.impl;

import cn.hutool.core.util.StrUtil;
import org.springframework.stereotype.Component;
import top.mddata.base.exception.ArgumentException;
import top.mddata.base.utils.ArgumentAssert;
import top.mddata.console.dto.coderule.CodeRuleItemConfig;
import top.mddata.console.enumeration.coderule.CodeRuleTypeEnum;
import top.mddata.console.enumeration.coderule.OverflowStrategyEnum;
import top.mddata.console.service.coderule.generator.GenerateContext;
import top.mddata.console.service.coderule.generator.RuleItemGenerator;

/**
 * 流水号生成器：从上下文取预分配值（引擎批量时已用 INCRBY 一次原子分配连续区间），按长度补位/超长处理。
 */
@Component
public class SerialRuleItemGenerator implements RuleItemGenerator {

    @Override
    public CodeRuleTypeEnum getType() {
        return CodeRuleTypeEnum.SERIAL;
    }

    @Override
    public String generate(CodeRuleItemConfig config, GenerateContext context) {
        Long value = context.nextSerial();
        ArgumentAssert.notNull(value, "流水号分配失败");
        return padAndCheck(value, config);
    }

    /**
     * 补位与超长处理：数字位数 < length 时按 padChar（默认 0）左补位；超过 length 时按超长策略报错或原样输出
     */
    public static String padAndCheck(long value, CodeRuleItemConfig config) {
        String raw = String.valueOf(value);
        int length = config.getLength() == null ? 0 : config.getLength();
        if (length <= 0 || raw.length() == length) {
            return raw;
        }
        if (raw.length() < length) {
            char pad = StrUtil.isBlank(config.getPadChar()) ? '0' : config.getPadChar().charAt(0);
            return StrUtil.padPre(raw, length, pad);
        }
        if (OverflowStrategyEnum.GROW.getCode().equals(config.getOverflowStrategy())) {
            return raw;
        }
        throw new ArgumentException("流水号[{}]已超过配置长度[{}]，当前策略为报错", raw, length);
    }
}
```

```java
import top.mddata.console.dto.coderule.CodeRuleItemConfig;
import top.mddata.console.enumeration.coderule.CheckDigitAlgorithmEnum;
import top.mddata.console.enumeration.coderule.CodeRuleTypeEnum;
import top.mddata.console.service.coderule.generator.GenerateContext;
import top.mddata.console.service.coderule.generator.RuleItemGenerator;

/**
 * 校验位生成器：基于已生成前文计算校验位（Luhn 仅支持纯数字前文，含字母自动用 Luhn Mod N）。
 * 防伪造、防录入错误。
 */
@Component
public class CheckDigitRuleItemGenerator implements RuleItemGenerator {

    private static final String LUHN_MOD_N_CHARS = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ";

    @Override
    public CodeRuleTypeEnum getType() {
        return CodeRuleTypeEnum.CHECK_DIGIT;
    }

    @Override
    public String generate(CodeRuleItemConfig config, GenerateContext context) {
        String prefix = context.getPrefix().toString();
        ArgumentAssert.notBlank(prefix, "校验位规则不能作为第一段（无前文可校验）");
        boolean allDigit = prefix.chars().allMatch(Character::isDigit);
        if (allDigit && !CheckDigitAlgorithmEnum.LUHN_MOD_N.getCode().equals(config.getAlgorithm())) {
            return String.valueOf(luhn(prefix));
        }
        return String.valueOf(luhnModN(prefix));
    }

    /**
     * 标准 Luhn 校验位（纯数字）
     */
    static int luhn(String digits) {
        int sum = 0;
        boolean doubleIt = true;
        for (int i = digits.length() - 1; i >= 0; i--) {
            int d = digits.charAt(i) - '0';
            if (doubleIt) {
                d = d * 2;
                if (d > 9) {
                    d -= 9;
                }
            }
            sum += d;
            doubleIt = !doubleIt;
        }
        return (10 - sum % 10) % 10;
    }

    /**
     * Luhn Mod N（字母数字，字母按 A=10 起映射）
     */
    static char luhnModN(String text) {
        int n = LUHN_MOD_N_CHARS.length();
        int sum = 0;
        boolean doubleIt = true;
        for (int i = text.length() - 1; i >= 0; i--) {
            int v = LUHN_MOD_N_CHARS.indexOf(Character.toUpperCase(text.charAt(i)));
            if (v < 0) {
                continue;
            }
            if (doubleIt) {
                v *= 2;
                if (v >= n) {
                    v = v - n + 1;
                }
            }
            sum += v;
            doubleIt = !doubleIt;
        }
        int check = (n - sum % n) % n;
        return LUHN_MOD_N_CHARS.charAt(check);
    }
}
```

```java
package top.mddata.console.service.coderule.generator.impl;

import org.springframework.stereotype.Component;
import top.mddata.base.exception.ArgumentException;
import top.mddata.base.utils.ArgumentAssert;
import top.mddata.console.dto.coderule.CodeRuleItemConfig;
import top.mddata.console.enumeration.coderule.CodeRuleTypeEnum;
import top.mddata.console.enumeration.coderule.CutTypeEnum;
import top.mddata.console.service.coderule.generator.GenerateContext;
import top.mddata.console.service.coderule.generator.RuleItemGenerator;

/**
 * 父单号截取生成器：从调用方 map 取父单号并截取复用。
 * INDEX：按下标截取（前缀定长）；SEPARATOR：取第 N 个分隔符之后的全部（支持负数从后往前数）。
 */
@Component
public class ParentCutRuleItemGenerator implements RuleItemGenerator {

    @Override
    public CodeRuleTypeEnum getType() {
        return CodeRuleTypeEnum.PARENT_CUT;
    }

    @Override
    public String generate(CodeRuleItemConfig config, GenerateContext context) {
        ArgumentAssert.notBlank(config.getVarName(), "父单号截取规则必须配置变量名");
        String parentCode = context.getParams().get(config.getVarName());
        ArgumentAssert.notNull(parentCode, "缺少变量[{}]", config.getVarName());
        if (CutTypeEnum.SEPARATOR.getCode().equals(config.getCutType())) {
            return cutBySeparator(parentCode, config);
        }
        return cutByIndex(parentCode, config);
    }

    private String cutByIndex(String parentCode, CodeRuleItemConfig config) {
        int startIndex = config.getStartIndex() == null ? 0 : config.getStartIndex();
        if (startIndex < 0 || startIndex > parentCode.length()) {
            throw new ArgumentException("父单号[{}]长度不足，无法从下标[{}]截取", parentCode, startIndex);
        }
        Integer length = config.getLength();
        int end = (length == null || length <= 0) ? parentCode.length()
                : Math.min(startIndex + length, parentCode.length());
        return parentCode.substring(startIndex, end);
    }

    /**
     * 取第 splitPosition 个分隔符之后的全部内容；负数从后往前数（-1=最后一个分隔符）
     */
    private String cutBySeparator(String parentCode, CodeRuleItemConfig config) {
        String separator = config.getSeparator();
        ArgumentAssert.notBlank(separator, "父单号截取规则必须配置分隔符");
        int splitPosition = config.getSplitPosition() == null ? 1 : config.getSplitPosition();
        if (splitPosition == 0) {
            throw new ArgumentException("分隔符位置不能为 0");
        }
        int fromIndex = 0;
        if (splitPosition > 0) {
            int pos = -1;
            for (int i = 0; i < splitPosition; i++) {
                pos = parentCode.indexOf(separator, pos + 1);
                if (pos < 0) {
                    throw new ArgumentException("父单号[{}]中不存在第[{}]个分隔符[{}]", parentCode, splitPosition, separator);
                }
            }
            fromIndex = pos + separator.length();
        } else {
            int pos = parentCode.length();
            for (int i = 0; i < -splitPosition; i++) {
                pos = parentCode.lastIndexOf(separator, pos - 1);
                if (pos < 0) {
                    throw new ArgumentException("父单号[{}]中不存在倒数第[{}]个分隔符[{}]", parentCode, -splitPosition, separator);
                }
            }
            fromIndex = pos + separator.length();
        }
        return parentCode.substring(fromIndex);
    }
}
```

```java
    private List<String> doGenerate(String ruleCode, List<Map<String, String>> paramsList, boolean preview) {
        // 1. 加载规则缓存（cacheOps.get(key, loader)）；不存在/禁用/无启用明细 → ArgumentAssert 报错
        CodeRuleCache cache = loadRule(ruleCode);

        // 2. 无流水号/GUID 明细时告警（唯一性无保障，仅警告不拦截）
        boolean hasUniqueSegment = cache.getItems().stream().anyMatch(item ->
                item.getRuleType().equals(CodeRuleTypeEnum.SERIAL.getCode())
                || item.getRuleType().equals(CodeRuleTypeEnum.GUID.getCode()));
        if (!hasUniqueSegment) {
            log.warn("规则[{}]不含流水号/GUID明细，单号唯一性无法保证", ruleCode);
        }

        int count = paramsList.size();
        LocalDateTime bizTime = LocalDateTime.now();
        Long userId = ContextUtil.getUserId();

        // 3. 流水号段预分配：每个流水号明细一次性 INCRBY count 原子分配连续区间
        //    preview 模式不真实递增：读当前值模拟（peek）
        Map<Long, List<Long>> serialBlocks = allocateSerialBlocks(cache.getItems(), count, bizTime, preview);

        // 4. 逐参数 × 逐明细 生成拼接
        List<String> result = new ArrayList<>(count);
        for (int i = 0; i < count; i++) {
            GenerateContext ctx = new GenerateContext(ruleCode, paramsList.get(i), bizTime, userId);
            int batchIndex = i;
            StringBuilder code = new StringBuilder();
            for (CodeRuleItem item : cache.getItems()) {
                ctx.setSerialSupplier(itemId -> serialBlocks.get(itemId).get(batchIndex));
                RuleItemGenerator generator = generatorRegistry.get(CodeRuleTypeEnum.valueOf(...item.getRuleType()...));
                String segment = generator.generate(item.getConfigJson(), ctx);
                ctx.appendPrefix(segment);
                code.append(segment);
            }
            result.add(code.toString());
        }
        return result;
    }
```
